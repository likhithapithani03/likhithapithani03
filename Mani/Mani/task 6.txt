#include <stdio.h>

int main() 
{
    int n, i;
    float sum = 0, avg;
    
    // Ask user for size
    printf("Enter number of elements: ");
    scanf("%d", &n);

    int arr[n];  // variable length array (C99 feature)

    // Input elements
    printf("Enter %d elements:\n", n);
    for (i = 0; i < n; i++) {
        scanf("%d", &arr[i]);
        sum += arr[i];  // accumulate sum
    }

    // Calculate average
    avg = sum / n;

    // Find largest and smallest
    int largest = arr[0];
    int smallest = arr[0];

    for (i = 1; i < n; i++) {
        if (arr[i] > largest)
            largest = arr[i];
        if (arr[i] < smallest)
            smallest = arr[i];
    }

    // Output results
    printf("\n--- Array Results ---\n");
    printf("Sum      : %.2f\n", sum);
    printf("Average  : %.2f\n", avg);
    printf("Largest  : %d\n", largest);
    printf("Smallest : %d\n", smallest);

    return 0;
}
